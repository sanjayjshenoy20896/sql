3.) SECOND highest salary for department 101
-> MAX aggregate func
-> group by
-> Having

CREATE TABLE employees (
    EmployeeID INT PRIMARY KEY,
    DepartmentID INT,
    Salary DECIMAL(10,2)
)

-> SUBQUERY

6.) Finding the total quantity for each product from stores in LA during 2023
Products structure:
ProductId INT PRIMARY KEY
ProductName VARCHAR(100)
CategoryId INT

Stores structure:
StoreId INT PRIMARY KEY
StoreName VARCHAR(100)
CategoryId INT

Sales structure
SaleId INT PRIMARY KEY
ProductId INT
StoreId INT
Quantity INT
SaleDate Date

select 
SUM(s.Quantity) as total_quantity,
p.productId,
st.StoreName,
MONTH(s.sale_date)
from Sales s
LEFT JOIN Stores st ON St.StoreId = s.StoreID
LEFT JOIN Product p on p.ProductId = s.ProductId
group by
p.ProductId,
st.StoreName
SUM(s.Quantity),
MONTH(s.sale_date)
Having 
st.StoreName = 'LA'
and Month(s.sale_date) = '2023'
and  p.ProductId IN (<IDS OF LAPTOP AND TABLET>);
